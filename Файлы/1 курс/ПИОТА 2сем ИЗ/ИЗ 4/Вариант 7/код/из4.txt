struct markov {

int vvod(struct markov* ptr_m) {
    ifstream fin("INPUT.txt");
    int l1, l2, x;
    fin >> l1;
    for (int i = 0; i < l1; i++) {
        fin >> ptr_m[i].nach >> ptr_m[i].zam;
        ptr_m[i].endd = true;
    }
    fin >> l2;
    for (int i = 0; i < l2; i++) {
        fin >> x;
        ptr_m[x - 1].endd = false;
    }
    return l1;
}

bool proverka(string h) {
    for (int i = 0; i < h.size(); i++) {
        if (h[i] != 'a' && h[i] != 'b') { return false; }
    }
    return true;
}
