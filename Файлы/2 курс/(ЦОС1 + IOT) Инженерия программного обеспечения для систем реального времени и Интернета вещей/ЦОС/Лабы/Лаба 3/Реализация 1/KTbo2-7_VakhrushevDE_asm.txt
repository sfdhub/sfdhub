#include "def21060.h"

#define N 3 //Y-multiplyer
#define M 10 //Frame size
#define Y_SPORT_DIV 10 //divider for Y-input samples (факт., потом в коде -1 будет для регистра)


.segment/pm rst_irq;
	nop;
	jump start;
.endseg;

/* 
ДМА-каналы
Канал 	Буфер
0		RX0		X-samples
1		RX1		Y-samples
2		TX0		Z-output
*/

.section/pm dm_data;
//===========================================	
//raw X samples buffers and TCB-blocks	
//start input from first buffer	
.var rawX1[M];
.var rawX2[M];

.var xSamplesTcb0[8] = {0, 0, 0,
					rawX2,
					xSamplesTcb1 + 7,
					@rawX1,
					1,
					rawX1};
.var xSamplesTcb1[8] = {0, 0, 0,
					rawX1,
					xSamplesTcb0 + 7,
					@rawX2,
					1,
					rawX2};
					
.var rawXAddr[2] = {rawX1, rawX2};

//===========================================
//raw Y samples buffers and TCB blocks
//start input from first buffer	
.var rawY1[N * M];
.var rawY2[N * M];

.var ySamplesTcb0[8] = {0, 0, 0, 
					rawY2,
					ySamplesTcb1 + 7,
					@rawY1,
					1,
					rawY1};
.var ySamplesTcb1[8] = {0, 0, 0,
					rawY1,
					ySamplesTcb0 + 7,
					@rawY2,
					1,
					rawY2};
					
.var rawYAddr[2] = {rawY1, rawY2};

//эти массивы должны быть вместе
.var avgY1[M];
.var avgY2[M];

//===========================================
//Z samples buffers and TCB-blocks	
//start output from second buffer				
.var Z1[M];
.var Z2[M];

.var zSamplesTcb0[8] = {0, 0, 0,
					Z1,
					zSamplesTcb1 + 7,
					@Z2,
					1,
					Z2};
.var zSamplesTcb1[8] = {0, 0, 0,
					Z2,
					zSamplesTcb0 + 7,
					@Z1,
					1,
					Z1};
					
.var ZAddr[2] = {Z2, Z1};		
.endseg;

.section/pm irq_spr0;
	R0 = DM(I0, M0);
	jump xBufferHandler (db);
		I3 = R0;	//I3 - X-buffer ptr
		R0 = DM(I1, M0);
.endseg;


.section/pm pm_code;
start:
		BIT SET MODE1 TRUNCATE;

//buffers addres selectors
//Такой метод быстрее, чем вычисление из GP
		B0 = rawXAddr;
		L0 = 2;
		
		B1 = rawYAddr;
		L1 = 2;
		
		B2 = ZAddr;
		L2 = 2;
		
		B6 = avgY1;
		L6 = 2 * M;
		
		M0 = 1;
		
//DMA-settings for SPORT0 - TX, RX, SPORT1 - RX ============
		R0 = Y_SPORT_DIV - 1;
 		DM(RDIV1) = R0;	//div for Y-samples
 		
 		R0 = (Y_SPORT_DIV * N) - 1;
 		DM(RDIV0) = R0;	//div for X-samples
 		DM(TDIV0) = R0;	//div for Z-output
		
		
		// Setting's for SPORT0 - RX, TX and SPORT1 - RX
		R0 = 0xC05F1;		//SPEN = 1
							//SLEN = 31 (32 факт.)
							//ICLCK = 1
							//DEN, SCHEN = 1
							
 		DM(SRCTL0) = R0;	//X - RX0 
 		DM(SRCTL1) = R0;	//Y - RX1
 		DM(STCTL0) = R0;	//Z - TX0
 		
 		R0 = ySamplesTcb0 + 7;
 		DM(CP1) = R0;		//settings Y-input DMA
 		
 		R0 = xSamplesTcb0 + 7;
 		DM(CP0) = R0;		//settings X-input DMA
 		
 		R0 = zSamplesTcb0 + 7;
 		DM(CP2) = R0;		//settings Z-output DMA


		
		BIT SET IMASK SPR0I;	//enbale only X-DMA irq
		BIT SET MODE1 IRPTEN; 	//enable IRQ's
//===========================================================
sleep:	idle;
	jump sleep;
	
/*
RX0 DMA irq handler
Used:
	B0 - xAddr, B1 - yAddr, B2 - zAddr, M0 = 1
	rawX(1/2), rawY(1/2) - input buffers after DMA, Z(1/2) - output buff for DMA 
Tempory used:
	R0, R1, R2, R3
	I3, I4, I5
Result: calculation result of input packet - Z(1/2)
*/
xBufferHandler:
	I4 = R0;	// I4 - Y-buffer ptr
	
	R0 = DM(I2, M0);
	I5 = R0;	// I5 - Z-buffer ptr
	
	//I3 - x, I4 - Y, I5 - Z
	
	F3 = 1.0 / N;	//for calc avg Y (Y`)
	
	LCNTR = M, DO frameHandler UNTIL LCE;
	
		R0 = 0;	//acc
		R1 = DM(I4, M0);
		

		#if N >= 4	//LCNTR = 3
			LCNTR = N - 1, DO ySum UNTIL LCE;
				ySum: R0 = R0 + R1, R1 = DM(I4, M0);
		#elif N == 3
			R0 = R0 + R1, R1 = DM(I4, M0);
			R0 = R0 + R1, R1 = DM(I4, M0);
		#elif N == 2
			R0 = R0 + R1, R1 = DM(I4, M0);
		#endif
		
		
		//Calc avg y 
		R0 = R0 + R1;
		F0 = FLOAT R0;
		F0 = F0 * F3;
		
		//round avg, read x
		R0 = FIX F0, R1 = DM(I3, M0);
		
		
		R2 = (R0 + R1) / 2, DM(I6, M0) = R0;//Z = (y` + x) / 2
frameHandler:DM(I5, M0) = R2; 	//save result
	rti;
