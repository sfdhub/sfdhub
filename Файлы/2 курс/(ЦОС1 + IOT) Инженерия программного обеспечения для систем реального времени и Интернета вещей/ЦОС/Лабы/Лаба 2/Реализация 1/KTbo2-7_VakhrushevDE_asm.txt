#include "def21060.h"

#define N 2

.segment/pm rst_irq;
	nop;
	jump start;
.endseg;

.segment/pm tmh_irq;
	jump sampleHandler (db);
		R13 = DM(in);
		R12 = R13 - R15;
.endseg;


.section/pm dm_data;
.var template[N] = "v3_template1.dat";
.var sign[N-1];
.endseg;

.section/pm pm_data;
.var inDataBuff[N-1];
.endseg;

.section/pm dm_port;
	.var in;
	.var out;
.endseg;
	


.section/pm pm_code;
start:
	B0 = sign;
	L0 = @sign;
	M0 = 1;
	M1 = -1;
	
	I1 = template;
	R14 = 1 << 31;
	LCNTR = N - 1, DO signCalc UNTIL LCE;
		R0 = DM(I1, M0);
		R1 = DM(I1, M7);
		R0 = R1 - R0;
		R0 = R0 and R14;
	signCalc:DM(I0, M0) = R0;
	
	
	
	B9 = inDataBuff;
	L9 = @inDataBuff;
	M8 = 1;
	
	TPERIOD = 100;

	R1 = 0;
	bit set MODE1 IRPTEN;
	bit set MODE2 TIMEN;
	bit set IMASK TMZHI;
	
sleep:	idle;
	jump sleep;

	
/*
Название: обработка прерывания по таймеру
Получает:
	in - input sample
	I9 - inDataBuffer
	I0 - sign Buffer
Портит:
	R15, R14, R13, R12, R11, R10, R9
Результат:
	out - результат
*/
sampleHandler:
	R15 = R13;
	R12 = R12 and R14;
	PM(I9, M15) = R12;
	
	R13 = 1;
	R9 = 0;
	LCNTR = N - 1, DO compareTemplate UNTIL LCE;
		R11 = DM(I0, M1), R12 = PM(I9, M8);
		COMP(R11, R12);
		compareTemplate:IF NE R13 = R9;

	
	rti (db);
	MODIFY(I9, M8);
	DM(out) = R13;
