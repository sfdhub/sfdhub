use [Hotel];

-- Обновляем таблицу "Reservation"
update [Reservation]

-- Устанавливаем значение столбца "TypeNumber" на основе условий
-- Если значение столбца "Cost" меньше 2000, то устанавливаем "Стандартный"
-- Если значение столбца "Cost" находится в диапазоне от 2000 до 3500, то устанавливаем "Улучшенный"
-- Если значение столбца "Cost" больше 3500, то устанавливаем "Люкс"
set [TypeNumber] = case 

	when [Cost] < 2000 then 'Стандартный'
	when [Cost] >= 2000 and [Cost] <= 3500 then 'Улучшенный'
	when [Cost] > 3500 then 'Люкс'

end

CREATE VIEW View_TotalDeptors AS
SELECT
    c.[IDClient],
    c.[Surname] + ' ' + c.[Name] + ' ' + c.[Patronymic] AS FIO,
    n.[Cost] * rn.[NumberDay] AS Cost,
    SUM(p.[Payment]) AS Payment
FROM
    [dbo].[Clients] c
    JOIN [dbo].[Accomodation] a ON c.IDClient = a.IDClient
    JOIN [PaymentCard] p ON p.[IDPaymentCard] = a.[IDPaymentCard]
    JOIN [dbo].[Reservation_Number] rn ON rn.[IDClient] = c.[IDClient]
    JOIN [dbo].[Numbers] n ON n.[IDNumber] = rn.[IDNumber]
WHERE
    a.[PayDate] IS NULL
GROUP BY
    c.[IDClient],
    c.[Surname] + ' ' + c.[Name] + ' ' + c.[Patronymic],
    n.[Cost] * rn.[NumberDay];

CREATE PROCEDURE Proc_NumbersStatistic
    @StartDay DATE,
    @EndDay DATE
AS
BEGIN
    SELECT n.[IDNumber]
        , COUNT(rn.[IDReserved]) AS CountNumber
        , ISNULL(SUM(n.[Cost]*rn.[NumberDay]), 0) AS TotalCost
        , ISNULL(SUM(rn.[NumberDay]), 0) AS TotalDays
    FROM [Hotel].[dbo].[Numbers] n
    LEFT JOIN [Hotel].[dbo].[Reservation_Number] rn ON n.[IDNumber] = rn.[IDNumber]
                                                 AND rn.[ArrivalDate] >= @StartDay
                                                 AND rn.[ArrivalDate] <= @EndDay
    GROUP BY n.[IDNumber]
    ORDER BY n.[IDNumber];
END;
